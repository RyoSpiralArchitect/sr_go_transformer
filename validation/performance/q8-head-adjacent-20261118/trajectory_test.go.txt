package main

import (
	"context"
	"crypto/sha256"
	"encoding/binary"
	"encoding/hex"
	"fmt"
	"hash"
	"reflect"
	"time"
)

type headTrajectoryBranch struct {
	caches []*KVCache
	chunks [][]int
	logits hash.Hash
}

type headTrajectorySample struct {
	Round      int        `json:"round"`
	Comparison int        `json:"comparison"`
	Modes      [2]string  `json:"modes"`
	First      []int      `json:"first_arm"`
	Elapsed    [2][]int64 `json:"elapsed_ns"`
	Head       [2][]int64 `json:"head_ns"`
	Dispatches [2]int     `json:"tiled_dispatches"`
	Signature  [2]string  `json:"logits_sha256"`
	KV         [2]string  `json:"kv_sha256"`
}

func trajectoryKVDigest(caches []*KVCache) string {
	h := sha256.New()
	buf := make([]byte, 0, 4*caches[0].Model.Config.KVDim())
	for _, cache := range caches {
		for layer := range cache.Model.Blocks {
			for pos := 0; pos < cache.Pos; pos++ {
				k, v := cache.rowF16(layer, pos)
				buf = buf[:0]
				for _, values := range [2][]uint16{k, v} {
					for _, value := range values {
						buf = binary.LittleEndian.AppendUint16(buf, value)
					}
				}
				h.Write(buf)
			}
		}
	}
	return hex.EncodeToString(h.Sum(nil))
}

func measureHeadTrajectories(ctx context.Context, m *Model, e *kernelExecutor, workspace *decodeWorkspace, prefixes []*KVCache, first []float32, reference contextSweepDecode) (any, error) {
	if len(prefixes) != 2 || prefixes[0].format != "f16" || prefixes[1].format != "f16" {
		return nil, fmt.Errorf("expected two FP16 sessions")
	}
	branches := [2]headTrajectoryBranch{}
	for j := range branches {
		branch := &branches[j]
		branch.chunks = make([][]int, len(prefixes))
		branch.logits = sha256.New()
		for _, prefix := range prefixes {
			cache, err := forkContextSweepCache(prefix)
			if err != nil {
				return nil, err
			}
			defer cache.Close()
			cache.ensurePages(prefix.Pos + contextSweepSteps)
			branch.caches = append(branch.caches, cache)
		}
	}

	run := func(round, comparison int, modes [2]string) (headTrajectorySample, error) {
		sample := headTrajectorySample{Round: round, Comparison: comparison, Modes: modes}
		timing := [2]adjacentHeadContext{{mode: modes[0]}, {mode: modes[1]}}
		probeContexts := [2]context.Context{
			context.WithValue(ctx, adjacentHeadKey{}, &timing[0]),
			context.WithValue(ctx, adjacentHeadKey{}, &timing[1]),
		}
		ids := [2][][]int{}
		for j := range branches {
			branch := &branches[j]
			branch.logits.Reset()
			perfHashFloats(branch.logits, first)
			ids[j] = make([][]int, len(prefixes))
			for session, cache := range branch.caches {
				cache.Pos = prefixes[session].Pos
				ids[j][session] = append(ids[j][session], reference.IDs[session][0])
			}
		}
		for step := 0; step < contextSweepSteps; step++ {
			firstArm := (round + step + comparison) % 2
			sample.First = append(sample.First, firstArm)
			for _, j := range [2]int{firstArm, 1 - firstArm} {
				branch := &branches[j]
				for session := range branch.chunks {
					branch.chunks[session] = ids[j][session][step : step+1]
				}
				timing[j].headNS, timing[j].calls = 0, 0
				count := adjacentTiledCalls
				start := time.Now()
				logits, err := m.prefillBorrowed(probeContexts[j], branch.caches, branch.chunks, e, workspace)
				elapsed := time.Since(start).Nanoseconds()
				if err != nil {
					return sample, err
				}
				if timing[j].calls != 1 || workspace.floatBytes() != reference.Workspace {
					return sample, fmt.Errorf("%s head count or workspace changed", modes[j])
				}
				sample.Elapsed[j] = append(sample.Elapsed[j], elapsed)
				sample.Head[j] = append(sample.Head[j], timing[j].headNS)
				sample.Dispatches[j] += adjacentTiledCalls - count
				perfHashFloats(branch.logits, logits)
				for session := range branch.caches {
					row := logits[session*m.Config.Vocab : (session+1)*m.Config.Vocab]
					best := 0
					for index, value := range row {
						if !finite(float64(value)) {
							return sample, fmt.Errorf("nonfinite logit in %s", modes[j])
						}
						if value > row[best] {
							best = index
						}
					}
					ids[j][session] = append(ids[j][session], best)
					if branch.caches[session].Pos != prefixes[session].Pos+step+1 {
						return sample, fmt.Errorf("position changed")
					}
				}
			}
		}
		for j := range branches {
			branch := &branches[j]
			sample.Signature[j] = hex.EncodeToString(branch.logits.Sum(nil))
			sample.KV[j] = trajectoryKVDigest(branch.caches)
			if sample.Signature[j] != reference.Logits || sample.KV[j] != reference.KV ||
				!reflect.DeepEqual(ids[j], reference.IDs) {
				return sample, fmt.Errorf("%s trajectory differed", modes[j])
			}
		}
		return sample, nil
	}
	for _, mode := range []string{"reference", "pair", "triple"} {
		if _, err := run(0, 0, [2]string{mode, mode}); err != nil {
			return nil, err
		}
	}

	var samples []headTrajectorySample
	for round := 0; round < trajectoryRounds; round++ {
		for _, slot := range trajectoryOrderSeed {
			index := (slot + round) % len(trajectoryComparisons)
			sample, err := run(round, index, trajectoryComparisons[index])
			samples = append(samples, sample)
			if err != nil {
				return samples, err
			}
		}
	}
	return samples, nil
}
