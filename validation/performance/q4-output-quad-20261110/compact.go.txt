// One activation block feeds four independent outputs. Each output accumulates
// the same four column lanes as dotQ4OutputPair64.
func dotQ4OutputQuad64(x *[64]float32, w0, w1, w2, w3 *[32]byte) (float32, float32, float32, float32) {
	var a0, b0, c0, d0, a1, b1, c1, d1 float32
	var a2, b2, c2, d2, a3, b3, c3, d3 float32
	for j := 0; j < 16; j++ {
		p0, p1 := q4Pairs[w0[j*2]], q4Pairs[w1[j*2]]
		p2, p3 := q4Pairs[w2[j*2]], q4Pairs[w3[j*2]]
		x0, x1 := x[j*4], x[j*4+1]
		a0 += x0 * p0[0]
		a1 += x0 * p1[0]
		a2 += x0 * p2[0]
		a3 += x0 * p3[0]
		b0 += x1 * p0[1]
		b1 += x1 * p1[1]
		b2 += x1 * p2[1]
		b3 += x1 * p3[1]

		q0, q1 := q4Pairs[w0[j*2+1]], q4Pairs[w1[j*2+1]]
		q2, q3 := q4Pairs[w2[j*2+1]], q4Pairs[w3[j*2+1]]
		x2, x3 := x[j*4+2], x[j*4+3]
		c0 += x2 * q0[0]
		c1 += x2 * q1[0]
		c2 += x2 * q2[0]
		c3 += x2 * q3[0]
		d0 += x3 * q0[1]
		d1 += x3 * q1[1]
		d2 += x3 * q2[1]
		d3 += x3 * q3[1]
	}
	return (a0 + b0) + (c0 + d0), (a1 + b1) + (c1 + d1), (a2 + b2) + (c2 + d2), (a3 + b3) + (c3 + d3)
}

// Applies to the same one-row, complete-block workload as the paired path.
func linearQ4OutputQuadRange(job *fp32Kernel, lo, hi int) {
	x, y, w := job.x, job.y, job.w
	groups, stride := w.Cols/64, w.Cols/2
	for tile := lo; tile < hi; tile++ {
		end := min((tile+1)*32, w.Rows)
		o := tile * 32
		for ; o+3 < end; o += 4 {
			w0 := w.Q4.Values[o*stride : (o+1)*stride]
			w1 := w.Q4.Values[(o+1)*stride : (o+2)*stride]
			w2 := w.Q4.Values[(o+2)*stride : (o+3)*stride]
			w3 := w.Q4.Values[(o+3)*stride : (o+4)*stride]
			sc0 := w.Q4.Scales[o*groups : (o+1)*groups]
			sc1 := w.Q4.Scales[(o+1)*groups : (o+2)*groups]
			sc2 := w.Q4.Scales[(o+2)*groups : (o+3)*groups]
			sc3 := w.Q4.Scales[(o+3)*groups : (o+4)*groups]
			var a, b, c, d float32
			for g := 0; g < groups; g++ {
				block := (*[64]float32)(x[g*64 : (g+1)*64])
				start, stop := g*32, (g+1)*32
				p, q, r, s := dotQ4OutputQuad64(block, (*[32]byte)(w0[start:stop]), (*[32]byte)(w1[start:stop]), (*[32]byte)(w2[start:stop]), (*[32]byte)(w3[start:stop]))
				a += p * sc0[g]
				b += q * sc1[g]
				c += r * sc2[g]
				d += s * sc3[g]
			}
			y[o], y[o+1], y[o+2], y[o+3] = a, b, c, d
		}
		// Rebase just the short suffix so the existing pair kernel owns tails.
		if o < end {
			param := Param{Cols: w.Cols, Rows: end - o, Q4: &q4Weights{
				Values: w.Q4.Values[o*stride : end*stride],
				Scales: w.Q4.Scales[o*groups : end*groups],
			}}
			tail := fp32Kernel{x: x, y: y[o:end], w: &param, n: 1}
			linearQ4OutputPairRange(&tail, 0, 1)
		}
	}
}

func TestQ4OutputQuadExact(t *testing.T) {
	for _, in := range []int{64, 192, 4096} {
		for _, out := range []int{1, 2, 3, 4, 31, 32, 33, 34, 35, 65, 67} {
			x, p, _ := linearFixture(1, in, out)
			w := quantizeQ4TestParam(t, p)
			want := make([]float32, out)
			guarded := make([]float32, out+2)
			guarded[0], guarded[out+1] = 123, 456
			tiles := (out + 31) / 32
			job := fp32Kernel{x: x, y: want, w: w, n: 1}
			linearQ4OutputPairRange(&job, 0, tiles)
			job.y = guarded[1 : out+1]
			e := newKernelExecutor(4)
			e.parallel(tiles, in*out, func(lo, hi int) { linearQ4OutputQuadRange(&job, lo, hi) })
			e.close()
			exactWorkerFloats(t, job.y, want)
			if guarded[0] != 123 || guarded[out+1] != 456 {
				t.Fatal("output overrun")
			}
		}
	}
}

func BenchmarkQ4OutputQuad(b *testing.B) {
	shapes := [][2]int{{4096, 1024}, {4096, 4096}, {4096, 14336}, {14336, 4096}}
	for _, shape := range shapes {
		in, out := shape[0], shape[1]
		x, p, _ := linearFixture(1, in, out)
		w := quantizeQ4TestParam(b, p)
		y := make([]float32, out)
		job := fp32Kernel{x: x, y: y, w: w, n: 1}
		for _, workers := range []int{1, 4} {
			e := newKernelExecutor(workers)
			tiles := (out + 31) / 32
			for sample, quad := range []bool{false, true, true, false} {
				name := "pair"
				fn := func(lo, hi int) { linearQ4OutputPairRange(&job, lo, hi) }
				if quad {
					name = "quad"
					fn = func(lo, hi int) { linearQ4OutputQuadRange(&job, lo, hi) }
				}
				b.Run(fmt.Sprintf("in%d/out%d/w%d/sample%d/%s", in, out, workers, sample, name), func(b *testing.B) {
					b.ReportAllocs()
					for i := 0; i < b.N; i++ {
						e.parallel(tiles, in*out, fn)
					}
				})
			}
			e.close()
		}
	}
}
